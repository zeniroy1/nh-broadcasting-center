function isUtilityMenuAllowed_(){

    .map(email => String(email || '').trim().toLowerCase())
    .filter(Boolean);
  if (!allowed.length) return false;

  const emails = [];
  try { emails.push(Session.getActiveUser().getEmail()); } catch(e){}
  try { emails.push(Session.getEffectiveUser().getEmail()); } catch(e){}

  return emails
    .map(email => String(email || '').trim().toLowerCase())
    .some(email => email && allowed.includes(email));

function sync_today_only(){
  const YOIL = ['일','월','화','수','목','금','토'];
  const today = YOIL[new Date().getDay()];
  if (!CONFIG.SOURCE_SHEET_NAMES.includes(today)) {
    SpreadsheetApp.getUi().alert(`업무일지 탭은 평일만 존재합니다. (오늘: ${today})`);
    return;
  }
  sync_core_([today]);
}

function sync_core_(sheetNames){
  if (SpreadsheetApp.getActive().getId() === CONFIG.SOURCE_SPREADSHEET_ID) {
    throw new Error('이 코드는 2번 파일에서만 실행하세요. 1번 파일은 읽기 전용입니다.');
  }
  ensure_target_sheets_exist_();

  const srcSS = SpreadsheetApp.openById(CONFIG.SOURCE_SPREADSHEET_ID);

  // A열 구분 검증: 누락된 모든 오류를 일괄 수집하여 연동 즉시 취소 및 외부 클릭이 무력화된 모달창 표출
  if (!validate_source_sheets_division_(srcSS, sheetNames)) {
    return;
  }

  const stats = {};
  Object.keys(CONFIG.TARGET_SHEET_BY_BUILDING).forEach(k => stats[k] = {written:0, left:0});

  sheetNames.forEach(name => {
    const sh = srcSS.getSheetByName(name);
    if (!sh) return;
    const dateInfo = extract_date_from_sheet_(sh);
    if (!dateInfo) return;
    const {month, day} = dateInfo;

    const sections = read_sections_from_source_(sh); // {본관:[], 신관:[]}
    Object.keys(CONFIG.TARGET_SHEET_BY_BUILDING).forEach(b => {
      const rows = sections[b] || [];
      const {written, left} = upsert_one_day_noinsert_(b, month, day, rows);
      stats[b].written += written;
      stats[b].left    += left;
    });
  });

  Object.values(CONFIG.TARGET_SHEET_BY_BUILDING).forEach(apply_sheet_font_);
  showSyncReport_(stats);
}

function ensure_target_sheets_exist_(){
  const ss = SpreadsheetApp.getActive();
  Object.values(CONFIG.TARGET_SHEET_BY_BUILDING).forEach(n => {
    if (!ss.getSheetByName(n)) throw new Error('타깃 시트를 찾을 수 없습니다: ' + n);

}

function extract_date_from_sheet_(sh){
  const vals = sh.getRange(1, 1, Math.min(sh.getLastRow(), 20), Math.min(sh.getLastColumn(), 15)).getDisplayValues();
  for (let r = 0; r < vals.length; r++){
    for (let c = 0; c < vals[r].length; c++){
      const s = String(vals[r][c] || '');
      const m = s.match(/(\d{4})\s*년\s*(\d{1,2})\s*월\s*(\d{1,2})\s*일/);
      if (m) return {year:+m[1], month:+m[2], day:+m[3]};
    }
  }
  return null;
}

function splitSubjectAndTitle_(s){
  const first = firstLine_(s);
  const m = first.match(/\(([^()]*)\)\s*$/);
  if (m) return { subject: first.slice(0, m.index).trim(), title: m[1].trim() };
  return { subject: first, title: '' };
}

function read_sections_from_source_(sh){
  const out = {'본관':[], '신관':[]};
  const lastRow = sh.getLastRow();
  if (lastRow < 1) return out;

  const aRange = sh.getRange(1, 1, lastRow, 1);
  const aVals  = aRange.getDisplayValues().map(r => r[0] || '');
  const merges = aRange.getMergedRanges();

  const sections = [];
  merges.forEach(m => {
    if (m.getColumn() !== 1) return;
    const label = _tc(m.getDisplayValue());
    if (label === '본관' || label === '신관')
      sections.push({type:label, start:m.getRow(), end:m.getRow()+m.getNumRows()-1});
  });

  function isInsideAny(r){ return sections.some(s => r >= s.start && r <= s.end); }
  for (let r = 1; r <= lastRow; r++){
    const label = _tc(aVals[r-1]);
    if ((label === '본관' || label === '신관') && !isInsideAny(r)){
      let end = lastRow;
      for (let rr = r + 1; rr <= lastRow; rr++){
        const l2 = _tc(aVals[rr-1]);
        if (l2 === '본관' || l2 === '신관'){ end = rr - 1; break; }
      }
      sections.push({type:label, start:r, end});
    }
  }

  sections.sort((a,b) => (a.start - b.start) || (a.end - b.end));
  const uniq = [];
  for (const s of sections){
    if (!uniq.length || uniq[uniq.length-1].start !== s.start || uniq[uniq.length-1].end !== s.end) uniq.push(s);
  }

  const {TIME:C, PLACE:D, SUBJECT:E, DEPT:F, HEAD:G} = CONFIG.SRC_COL;
  const headerTokens = new Set(['시간','운영시간','장소','회의실명','회의내용','회의명','주관부서','주관팀','담당','인원','비고','주요안내사항']);

  uniq.forEach(sec => {
    const n = Math.max(0, sec.end - sec.start + 1); if (!n) return;
    const rows = sh.getRange(sec.start, 1, n, Math.max(G, 7)).getDisplayValues();

    let lastDept = '';
    const seen = new Set();
    for (let i = 0; i < n; i++){
      const r = rows[i];
      const time = _t(r[C-1]);
      const place = _t(r[D-1]);
      const subjRaw = r[E-1];
      const {subject, title} = splitSubjectAndTitle_(subjRaw);
      let dept = _t(r[F-1]);
      const head = _t(r[G-1]);

      if (!dept) dept = lastDept; else lastDept = dept;

      if ([time,place,subject,dept,head].every(v => !v)) continue;
      if ([time,place,subject,dept,head].some(x => headerTokens.has(_tc(x)))) continue;
      if (!TIME_RE.test(time) || (!place && !subject)) continue;

      const key = [time, place, subject, dept].map(normKey_).join('|');
      if (seen.has(key)) continue; seen.add(key);

      out[sec.type].push({time, place, subject, title, dept, head});
    }
  });

  return out;
}

function format_headcount_(s){
  const n = String(s || '').replace(/[^\d]/g,'');
  return n ? n + '명' : '';
}

function lastDataRowBJ_(sh){
  const hr = CONFIG.TARGET_HEADER_ROWS, max = sh.getLastRow();
  if (max <= hr) return hr;
  const vals = sh.getRange(hr+1, 2, max-hr, 9).getValues();
  for (let i = vals.length-1; i >= 0; i--){
    if (vals[i].some(v => String(v).trim() !== '')) return hr + 1 + i;
  }
  return hr;
}

function copyTemplateFormatDJ_(sh, startRow, numRows){
  if (numRows <= 0) return;
  const tplRow = Math.min(sh.getLastRow(), CONFIG.TARGET_HEADER_ROWS + 1) || (CONFIG.TARGET_HEADER_ROWS + 1);
  const src = sh.getRange(tplRow, 4, 1, 7);
  const dst = sh.getRange(startRow, 4, numRows, 7);
  src.copyTo(dst, {formatOnly:true});
}

function upsert_one_day_noinsert_(building, month, day, rows){
  const name = CONFIG.TARGET_SHEET_BY_BUILDING[building];
  if (!name) return {written:0, left:rows.length};
  const sh = SpreadsheetApp.getActive().getSheetByName(name);
  if (!sh) throw new Error('타깃 시트를 찾지 못했습니다: ' + name);
  const hr = CONFIG.TARGET_HEADER_ROWS;

  const dataLast = lastDataRowBJ_(sh);
  const haveIdx = [];
  if (dataLast > hr){
    const bc = sh.getRange(hr+1, 2, dataLast-hr, 2).getValues();
    for (let i = 0; i < bc.length; i++){
      const m = Number(String(bc[i][0]).replace(/[^\d]/g,''));
      const d = Number(String(bc[i][1]).replace(/[^\d]/g,''));
      if (m === month && d === day) haveIdx.push(hr+1+i);
    }
  }
  
  const existingMap = new Map();
  haveIdx.forEach(idx => {
    const v = sh.getRange(idx, 4, 1, 4).getDisplayValues()[0];
    const k = [v[0], v[1], v[2], v[3]].map(normKey_).join('|');
    if (!existingMap.has(k)) existingMap.set(k, []);
    existingMap.get(k).push(idx);
  });

  const seen = new Set();
  const filtered = [];
  rows.forEach(r => {
    const k = [r.time, r.place, r.subject, r.dept].map(normKey_).join('|');
    if (!seen.has(k)) { seen.add(k); filtered.push(r); }
  });

  let written = 0;
  const used = new Set();
  const append = [];
  filtered.forEach(r => {
    const k = [r.time, r.place, r.subject, r.dept].map(normKey_).join('|');
    const targets = (existingMap.get(k) || []).filter(i => !used.has(i));
    const meetingType = r.title && r.title.trim() ? '주요회의' : '일반회의';
    const payload = [month, day, r.time, r.place, r.subject, r.dept, format_headcount_(r.head), r.title || '', meetingType];
    if (targets.length) {
      const idx = targets[0];
      sh.getRange(idx, 2, 1, 9).setValues([payload]);
      used.add(idx);
      written++;
    } else {
      append.push(payload);
    }
  });

  let left = 0;
  if (append.length){
    const start = lastDataRowBJ_(sh) + 1;
    const canWrite = Math.max(0, sh.getLastRow() - start + 1);
    const n = Math.min(canWrite, append.length);
    if (n > 0){
      copyTemplateFormatDJ_(sh, start, n);
      sh.getRange(start, 2, n, 9).setValues(append.slice(0, n));
      written += n;
    }
    left = append.length - n;
  }
  haveIdx.forEach(i => { if (!used.has(i)) sh.getRange(i, 2, 1, 9).clearContent(); });

  return {written, left};
}

function apply_sheet_font_(name){
  const sh = SpreadsheetApp.getActive().getSheetByName(name);
  if (!sh) return;

  const lr = Math.max(1, sh.getLastRow());
  const lc = Math.max(1, sh.getLastColumn());

  }

    sh.getRange(1, 11, 1, lc-10).setFontSize(CONFIG.FONT_SIZE);
  }
}

function photoFormat_cloneBlocksRowPairHere_(count){
  const sh = SpreadsheetApp.getActiveSheet(), ui = SpreadsheetApp.getUi(), sel = sh.getActiveRange();
  if (!sel) { ui.alert('먼저 복제할 블록의 한 칸을 선택해 주세요.'); return; }

  const slotName =
    (sh.getName().indexOf('본관') > -1 && sh.getName().indexOf('보수') > -1) ? PHOTO_INPUT.SLOT_NAME.MB :
    (sh.getName().indexOf('신관') > -1 && sh.getName().indexOf('보수') > -1) ? PHOTO_INPUT.SLOT_NAME.MS :
    (sh.getName().indexOf('본관') > -1) ? PHOTO_INPUT.SLOT_NAME.B :
    (sh.getName().indexOf('신관') > -1) ? PHOTO_INPUT.SLOT_NAME.S : null;

  if (!slotName) { ui.alert('이 기능은 본관/신관 (사진)·(보수점검) 시트에서만 사용하세요.'); return; }

  const base = sh.getParent().getRangeByName(slotName);
  if (!base || base.getSheet().getSheetId() !== sh.getSheetId()){ ui.alert('이름범위 "' + slotName + '" (첫 사진칸 1칸)부터 설정하세요.'); return; }
  if (base.getNumRows() !== 1 || base.getNumColumns() !== 1){ ui.alert('이름범위 "' + slotName + '" 은 1칸이어야 합니다.'); return; }

  const r0 = base.getRow(), cL = base.getColumn(), cR = cL + (PHOTO_INPUT.SECOND_COL_OFFSET || 2), STEP = 3;
  let top = sel.getRow(); top = top - ((top - r0) % STEP);
  if (LB_isHeaderRow_(sh, top, cL, cR)){ ui.alert('레터박스 행에서는 복제할 수 없습니다.'); return; }

  const tplL = sh.getRange(top, cL, STEP, 1), tplR = sh.getRange(top, cR, STEP, 1);
  const insertAfter = top + STEP - 1;
  sh.insertRowsAfter(insertAfter, STEP * count);

  for (let i = 0; i < count; i++){
    for (let r = 0; r < STEP; r++){
      const h = sh.getRowHeight(top+r);
      sh.setRowHeight(insertAfter + (i * STEP) + 1 + r, h);
    }
    const dstRow = insertAfter + 1 + (i * STEP);
    const dstL = sh.getRange(dstRow, cL, STEP, 1); tplL.copyTo(dstL, {formatOnly:true}); dstL.clearContent();
    const dstR = sh.getRange(dstRow, cR, STEP, 1); tplR.copyTo(dstR, {formatOnly:true}); dstR.clearContent();
  }

  PF_whitenColumnB_(sh, insertAfter+1, insertAfter + STEP * count);
  ui.alert(`블록을 아래로 ${count}개(각각 A/C 한 쌍) 추가했습니다.`);
}

function PF_whitenColumnB_(sheet, rowFrom, rowTo){
  if (rowTo < rowFrom) return;
  sheet.getRange(rowFrom, 2, rowTo-rowFrom+1, 1)
       .setBorder(true, null, true, null, null, null, COLORS.WHITE, SpreadsheetApp.BorderStyle.SOLID);
}

function PF_clearLetterboxBorders_(sheet, headerRow, spacerRow){
  sheet.getRange(headerRow, 1, 1, MONTH_HEADER.WIDTH_COLS)
    .setBorder(false, false, false, false, false, false);
  if (spacerRow){
    sheet.getRange(spacerRow, 1, 1, MONTH_HEADER.WIDTH_COLS)
      .setBorder(false, false, false, false, false, false);
  }
}

}

  });
}

function PF_collectPhotoSlot_(sheet, photoRow, col, photos){

function PF_copyTemplatePage_(sheet, templateSheet, startRow){
  const pageRows = PHOTO_DRYRUN.TEMPLATE_PAGE_ROWS;
  const maxCols = Math.min(sheet.getMaxColumns(), templateSheet.getMaxColumns());
  const src = templateSheet.getRange(1, 1, pageRows, maxCols);
  const dst = sheet.getRange(startRow, 1, pageRows, maxCols);

  src.copyTo(dst, SpreadsheetApp.CopyPasteType.PASTE_NORMAL, false);

  for (let i = 0; i < pageRows; i++){
    sheet.setRowHeight(startRow + i, templateSheet.getRowHeight(i + 1));
  }
  PF_clearLetterboxBorders_(sheet, startRow, startRow + 1);
  PF_restorePhotoFrameTopBorder_(sheet, startRow + PHOTO_DRYRUN.TEMPLATE_HEADER_ROWS);
}

function PF_fillTemplatePage_(sheet, startRow, headerText, photos){
  sheet.getRange(startRow, 1).setValue(headerText || MONTH_HEADER.MAIN_TEXT);

  for (let setIndex = 0; setIndex < PHOTO_DRYRUN.PAGE_SETS; setIndex++){
    const photoRow = startRow + PHOTO_DRYRUN.TEMPLATE_HEADER_ROWS + (setIndex * PHOTO_DRYRUN.STEP);
    const leftPhoto = photos[setIndex * 2];
    const rightPhoto = photos[(setIndex * 2) + 1];

    if (!leftPhoto && !rightPhoto){
      PF_clearTemplatePhotoSet_(sheet, photoRow);
    } else {
      PF_fillTemplatePhotoSlot_(sheet, photoRow, PHOTO_DRYRUN.LEFT_COL, leftPhoto);
      PF_fillTemplatePhotoSlot_(sheet, photoRow, PHOTO_DRYRUN.RIGHT_COL, rightPhoto);
    }
  }
}

function PF_fillTemplatePhotoSlot_(sheet, photoRow, col, photo){
  if (!photo){
    PF_clearTemplatePhotoSlot_(sheet, photoRow, col);
    return;
  }

  sheet.getRange(photoRow + 1, col).setValue(photo.title || '');
  sheet.getRange(photoRow + 2, col).setValue(photo.id || '');
}

function PF_getPhotoPagePlan_(sheet, selectedRow){
  const step = PHOTO_DRYRUN.STEP;
  const templateSheet = PF_getPhotoTemplateSheet_(sheet.getParent());
  if (!templateSheet){
    return {
      ok: false,
      message: '기준틀 시트 "' + PHOTO_DRYRUN.TEMPLATE_SHEET_NAME + '" 을 찾지 못했습니다.'
    };
  }

  const headerRow = PF_findPrevPhotoHeaderRow_(sheet, selectedRow);
  if (!headerRow){
    return {
      ok: false,
      message: '선택 행 위쪽에서 월 주요 회의/행사 레터박스를 찾지 못했습니다.'
    };
  }

  const firstPhotoRow = headerRow + 2;
  if (selectedRow < firstPhotoRow){
    return {
      ok: false,
      message: '선택 행이 레터박스 또는 빈행 영역입니다. 숨은 ID 입력행을 선택해 주세요.'
    };
  }

  if ((selectedRow - firstPhotoRow) % step !== step - 1){
    return {
      ok: false,
      message: '레터박스 삽입 기준은 숨은 ID 입력행입니다. 사진행/행사명행이 아니라 ID가 들어가는 행을 선택해 주세요.'
    };
  }

  const selectedSetIndex = Math.floor((selectedRow - firstPhotoRow) / step);
  const selectedSetTop = firstPhotoRow + (selectedSetIndex * step);
  const alignedInsertAfter = selectedSetTop + step - 1;
  const completedSets = Math.floor((alignedInsertAfter - firstPhotoRow + 1) / step);

  if (completedSets > PHOTO_DRYRUN.PAGE_SETS){
    return {
      ok: false,
      message: '사진세트 초과로 실행할 수 없습니다.\n\n레터박스 기준으로 1~3세트까지만 선택할 수 있습니다.\n현재 선택 위치는 ' + completedSets + '세트째입니다.\n\n형식 유지를 위해 실행을 중단했습니다.'
    };
  }

  const currentPageSets = completedSets % PHOTO_DRYRUN.PAGE_SETS;
  const missingSets = currentPageSets === 0 ? 0 : PHOTO_DRYRUN.PAGE_SETS - currentPageSets;
  const virtualRows = missingSets * step;
  const templateRowHeights = PF_getTemplateRowHeights_(templateSheet);
  const headerHeights = templateRowHeights.slice(0, PHOTO_DRYRUN.TEMPLATE_HEADER_ROWS);
  const rowHeights = templateRowHeights.slice(PHOTO_DRYRUN.TEMPLATE_HEADER_ROWS, PHOTO_DRYRUN.TEMPLATE_HEADER_ROWS + step);
  const templatePageHeight = templateRowHeights.reduce((sum, h) => sum + h, 0);
  const newHeaderRow = alignedInsertAfter + virtualRows + 1;
  const newSpacerRow = newHeaderRow + 1;
  const nextPhotoStartRow = newHeaderRow + 2;
  const sample = PF_describePhotoSet_(sheet, selectedSetTop);

  return {
    ok: true,
    headerRow,
    firstPhotoRow,
    selectedSetTop,
    alignedInsertAfter,
    completedSets,
    currentPageSets,
    missingSets,
    virtualRows,
    newHeaderRow,
    newSpacerRow,
    nextPhotoStartRow,
    rowHeights,
    headerHeights,
    templateSheet,
    templateName: templateSheet.getName(),
    templatePageHeight,
    sample
  };
}

function PF_getPhotoTemplateSheet_(spreadsheet){
  return spreadsheet.getSheetByName(PHOTO_DRYRUN.TEMPLATE_SHEET_NAME);
}

function PF_getTemplateRowHeights_(templateSheet){
  const heights = [];
  for (let row = 1; row <= PHOTO_DRYRUN.TEMPLATE_PAGE_ROWS; row++){
    heights.push(templateSheet.getRowHeight(row));
  }
  return heights;
}

function PF_applyTemplateColumnWidths_(sheet, templateSheet){
  const maxCols = Math.min(sheet.getMaxColumns(), templateSheet.getMaxColumns());
  for (let col = 1; col <= maxCols; col++){
    sheet.setColumnWidth(col, templateSheet.getColumnWidth(col));
  }
}

function PF_insertInvisiblePhotoSets_(sheet, startRow, setCount, rowHeights){
  const step = PHOTO_DRYRUN.STEP;
  const lastCol = Math.max(sheet.getLastColumn(), MONTH_HEADER.WIDTH_COLS);
  const rows = setCount * step;
  const range = sheet.getRange(startRow, 1, rows, lastCol);

  range.clearContent();
  range.clearFormat();
  range.setBackground(null);
  range.setBorder(false, false, false, false, false, false);

  for (let i = 0; i < rows; i++){
    sheet.setRowHeight(startRow + i, rowHeights[i % step]);
  }
}

function PF_createTemplateMonthHeader_(sheet, headerRow, spacerRow, templateSheet, headerHeights){
  sheet.setRowHeight(headerRow, headerHeights[0]);
  sheet.setRowHeight(spacerRow, headerHeights[1]);

  const headerRange = sheet.getRange(headerRow, 1, 1, MONTH_HEADER.WIDTH_COLS);
  const spacerRange = sheet.getRange(spacerRow, 1, 1, MONTH_HEADER.WIDTH_COLS);

  headerRange.breakApart();
  spacerRange.breakApart();
  headerRange.clear();
  spacerRange.clear();

  templateSheet.getRange(1, 1, 1, MONTH_HEADER.WIDTH_COLS).copyTo(headerRange, {formatOnly: true});

function LB_norm_(s){ return String(s || '').toLowerCase().replace(/\s+/g,'').replace(/[&/\\()]/g,''); }

function PUP_isLetterboxTopRow_(sh, row){
  const vals = sh.getRange(row, 1, 1, 3).getDisplayValues()[0].join(' ');
  const t = LB_norm_(vals);
  return t.includes(LB_norm_(MONTH_HEADER.MAIN_TEXT)) ||
         (t.includes('월보수') && (t.includes('장애예방') || t.includes('사용불능')));
}

function LB_isHeaderRow_(sheet, row, cL, cR){
  const cells = [sheet.getRange(row, cL), sheet.getRange(row, cR)];
  for (let i = 0; i < cells.length; i++){
    try {
      const cell = cells[i]; if (!cell.isPartOfMerge()) continue;
      const merged = cell.getMergedRanges()[0];
      if (merged && merged.getNumRows() === 1 && merged.getNumColumns() >= MONTH_HEADER.WIDTH_COLS){
        const t = LB_norm_(merged.getDisplayValue());
        if (t.includes(LB_norm_(MONTH_HEADER.MAIN_TEXT))) return true;
        if (t.includes('월보수') && (t.includes('장애예방') || t.includes('사용불능'))) return true;
      }
    } catch(_) {}
  }
  return false;
}

/* =========================================================
 * ▶ 사진업로드
 * ========================================================= */
function simplePhotoUpload_fillFromInput() {
  const sh = SpreadsheetApp.getActiveSheet(), ui = SpreadsheetApp.getUi();

  const idsRaw = sh.getRange('K1:N1').getValues()[0];
  const ids = idsRaw.map(v => PUP_extractDriveId(String(v || ''))).filter(Boolean);
  if (!ids.length){ ui.alert('K1:N1에 유효한 구글 링크 또는 ID가 없습니다.'); return; }

  const startFrom = PUP_forceBelowLetterbox_(sh, 2);
  let cursor = Math.max(3, startFrom);

  const placed = [];
  ids.forEach(id => {
    const slot = PUP_findNextFreeSlot_(sh, cursor); if (!slot) throw new Error('빈 슬롯을 찾지 못했습니다. 블록을 추가해 주세요.');

    const img = sh.getRange(slot.row, slot.col);
    const tit = sh.getRange(slot.row+1, slot.col);
    const idc = sh.getRange(slot.row+2, slot.col);

    idc.setValue(id).setFontColor(COLORS.WHITE);
    img.setFormula(`=IMAGE("https://drive.google.com/uc?export=download&id=" & ${idc.getA1Notation()}, 2)`);

    try { sheetResizeForPhoto_(sh); } catch(_) {}

    if (String(tit.getValue() || '').trim() === '') tit.setValue('행사 제목');

    placed.push(`${img.getA1Notation()} (ID=${id})`);
    cursor = slot.nextStart;
  });

  ui.alert(`총 ${placed.length}개 업로드 완료\n` + placed.join('\n'));
  sh.getRange('K1:N1').clearContent();
}

function sheetResizeForPhoto_(sh){
  try {
    sh.setColumnWidth(1, 280);
    sh.setColumnWidth(3, 280);
  } catch(_) {}
}

function PUP_forceBelowLetterbox_(sh, jump){
  const last = sh.getLastRow();
  for (let r = last; r >= 1; r--){
    if (PUP_isLetterboxTopRow_(sh, r)) return r + Math.max(2, jump | 0);
  }
  return 3;
}

function PUP_findNextFreeSlot_(sh, startRow){
  const lastRow = Math.max(sh.getMaxRows(), startRow);
  let r = Math.max(3, startRow); const colA = 1, colC = 3;
  while (r <= lastRow){
    if (PUP_isLetterboxTopRow_(sh, r)){ r += 2; continue; }
    if (!PUP_isBlockUsed_(sh, r, colA)) return {row:r, col:colA, nextStart:r};
    if (!PUP_isBlockUsed_(sh, r, colC)) return {row:r, col:colC, nextStart:r+3};
    r += 3;
  }
  return null;
}

function PUP_isBlockUsed_(sh, row, col){
  const idc = sh.getRange(row+2, col), img = sh.getRange(row, col), tit = sh.getRange(row+1, col);
  const idv = String(idc.getValue() || '').trim(), fm = String(img.getFormula() || '').trim(),
        dv = String(img.getDisplayValue() || '').trim(), tv = String(tit.getValue() || '').trim();
  const hasOverlay = (() => { try { return sh.getImages().some(im => {const a = im.getAnchorCell(); return a && a.getRow() === row && a.getColumn() === col;}); } catch(_) { return false; } })();
  return !!(idv || /^=IMAGE\(/i.test(fm) || dv || hasOverlay || tv);
}

function PUP_extractDriveId(str){
  if (!str) return '';
  let m = str.match(/\/d\/([A-Za-z0-9_-]{25,})/); if (m) return m[1];
  m = str.match(/[?&]id=([A-Za-z0-9_-]{25,})/); if (m) return m[1];
  return /^[A-Za-z0-9_-]{25,40}$/.test(str) ? str : '';
}

}
